```
import { describe, it, expect, beforeAll } from 'vitest';
import { JsonRpcProvider } from 'ethers';

describe('HONEY-03: HoneypotChecker deployed on ETH', () => {
    let provider: JsonRpcProvider;
    beforeAll(() => {
        if (!process.env.ETH_RPC_URL) throw new Error('ETH_RPC_URL required');
        provider = new JsonRpcProvider(process.env.ETH_RPC_URL);
    });
    it.skipIf(!process.env.ETH_HONEYPOT_CHECKER_ADDRESS && !__deployedAddressFromTemplate)(
      'has bytecode at deployedAddress (not 0x)',
      async () => {
        const deployedAddr = process.env.ETH_HONEYPOT_CHECKER_ADDRESS ?? __deployedAddressFromTemplate;
        const code = await provider.getCode(deployedAddr!);
        expect(code).not.toBe('0x');
        expect(code.length).toBeGreaterThan(100);
      });
});
```
