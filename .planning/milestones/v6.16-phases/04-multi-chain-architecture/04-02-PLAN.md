**`src/utils/mevStrategies/flashbots.ts`:**

Extract existing Flashbots logic from `txManager.ts` and `flashbots.ts` into a class implementing `MevStrategy`.

```typescript
import { JsonRpcProvider, Wallet, TransactionRequest } from 'ethers';
import { FlashbotsBundleProvider } from '@flashbots/ethers-provider-bundle';
import { MevStrategy } from './types';
import { logger } from '../logger';

const log = logger.child({ module: 'mev-flashbots' });

export class FlashbotsMevStrategy implements MevStrategy {
  readonly name = 'flashbots';
  private relayUrl: string;
  private network: string;
  private authKey: string;

  constructor(mevConfig: Record<string, string>) {
    this.relayUrl = mevConfig.relayUrl || 'https://relay.flashbots.net';
    this.network = mevConfig.network || 'mainnet';
    this.authKey = process.env.FLASHBOTS_AUTH_KEY!;
  }

  async healthCheck(): Promise<boolean> {
    // Verify Flashbots relay is reachable by creating a provider
    try {
      const { JsonRpcProvider, Wallet } = await import('ethers');
      const tempProvider = new JsonRpcProvider(this.relayUrl.replace('relay', 'rpc'));
      // Actually just verify the auth signer works and relay URL is valid format
      const authSigner = new Wallet(this.authKey);
      log.info({ relay: this.relayUrl }, 'Flashbots health check: auth key valid');
      return true;
    } catch (err) {
      log.error({ err }, 'Flashbots health check failed');
      return false;
    }
  }

  async sendTransaction(
    txRequest: TransactionRequest,
    wallet: Wallet,
    provider: JsonRpcProvider,
    maxRetries = 3
  ): Promise<string> {
    const authSigner = new Wallet(this.authKey);
    const flashbotsProvider = await FlashbotsBundleProvider.create(
      provider, authSigner, this.relayUrl, this.network
    );
    const baseNonce = await provider.getTransactionCount(wallet.address, 'latest');

    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        const gasPrice = (await provider.getFeeData()).gasPrice;
        const populatedTx: TransactionRequest = {
          ...txRequest,
          gasLimit: txRequest.gasLimit || 300_000n,
          gasPrice,
          nonce: baseNonce + attempt,
          chainId: (await provider.getNetwork()).chainId,
        };

        const targetBlock = (await provider.getBlockNumber()) + 1;
        const bundle = await flashbotsProvider.sendBundle(
          [{ signer: wallet, transaction: populatedTx }],
          targetBlock
        );

        if ('error' in bundle) {
          throw new Error(`Flashbots error: ${bundle.error.message}`);
        }

        const waitResponse = await bundle.wait();
        if (waitResponse === 0) {
          log.info({ block: targetBlock }, 'Flashbots bundle included');
          return `flashbots_bundle_block_${targetBlock}`;
        }
        throw new Error(`Bundle not included in block ${targetBlock}`);
      } catch (err) {
        log.warn({ attempt: attempt + 1, maxRetries, err }, 'Flashbots retry');
        if (attempt >= maxRetries) throw err;
      }
    }
    throw new Error('Flashbots max retries reached');
  }
}
```

**`src/utils/mevStrategies/bloxroute.ts`:**

Implement BSC MEV protection via bloXroute `bsc_private_tx` JSON-RPC (per D-05, research section).

```typescript
import { JsonRpcProvider, Wallet, TransactionRequest } from 'ethers';
import axios from 'axios';
import { MevStrategy } from './types';
import { logger } from '../logger';

const log = logger.child({ module: 'mev-bloxroute' });

export class BloxrouteMevStrategy implements MevStrategy {
  readonly name = 'bloxroute-bsc';
  private endpoint: string;
  private authHeader: string;

  constructor(mevConfig: Record<string, string>) {
    this.endpoint = mevConfig.endpoint || 'https://api.blxrbdn.com';
    this.authHeader = process.env.BLOXROUTE_AUTH_HEADER!;
  }

  async healthCheck(): Promise<boolean> {
    try {
      // Ping bloXroute with a benign JSON-RPC call
      const { data } = await axios.post(this.endpoint, {
        jsonrpc: '2.0', id: 1,
        method: 'ping',
        params: {}
      }, {
        headers: {
          'Content-Type': 'application/json',
          'Authorization': this.authHeader
        },
        timeout: 5000
      });
      // Any non-error response means service is reachable
      if (data.error && (data.error.code === 401 || data.error.code === 403)) {
        log.error({ error: data.error }, 'bloXroute auth failure');
        return false;
      }
      log.info('bloXroute health check passed');
      return true;
    } catch (err) {
      log.error({ err }, 'bloXroute health check failed');
      return false;
    }
  }

  async sendTransaction(
    txRequest: TransactionRequest,
    wallet: Wallet,
    provider: JsonRpcProvider,
    maxRetries = 3
  ): Promise<string> {
    const baseNonce = await provider.getTransactionCount(wallet.address, 'latest');

    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        const gasPrice = (await provider.getFeeData()).gasPrice;
        const populatedTx: TransactionRequest = {
          ...txRequest,
          gasLimit: txRequest.gasLimit || 500_000n,
          gasPrice,
          nonce: baseNonce + attempt,
          chainId: 56n,
        };

        const signedTx = await wallet.signTransaction(populatedTx);

        const { data } = await axios.post(this.endpoint, {
          jsonrpc: '2.0', id: 1,
          method: 'bsc_private_tx',
          params: { transaction: signedTx }
        }, {
          headers: {
            'Content-Type': 'application/json',
            'Authorization': this.authHeader
          },
          timeout: 10000
        });

        if (data.error) {
          throw new Error(`bloXroute error: ${data.error.message}`);
        }

        log.info({ txHash: data.result }, 'BSC private tx submitted via bloXroute');
        return data.result;
      } catch (err) {
        log.warn({ attempt: attempt + 1, maxRetries, err }, 'bloXroute retry');
        if (attempt >= maxRetries) throw err;
      }
    }
    throw new Error('bloXroute max retries reached');
  }
}
```

**`src/utils/mevStrategies/passthrough.ts`:**

Direct transaction submission for chains with natively private mempools (Base, Arbitrum). Per D-05 research: Base sequencer mempool is private, Arbitrum uses FCFS ordering.

```typescript
import { JsonRpcProvider, Wallet, TransactionRequest } from 'ethers';
import { MevStrategy } from './types';
import { logger } from '../logger';

const log = logger.child({ module: 'mev-passthrough' });

export class PassthroughMevStrategy implements MevStrategy {
  readonly name: string;

  constructor(chainName: string) {
    this.name = `passthrough-${chainName}`;
  }

  async healthCheck(): Promise<boolean> {
    // Passthrough chains (Base, Arbitrum) have natively private mempools.
    // No external MEV service to check -- always healthy.
    log.info({ strategy: this.name }, 'Passthrough MEV: always healthy (native private mempool)');
    return true;
  }

  async sendTransaction(
    txRequest: TransactionRequest,
    wallet: Wallet,
    provider: JsonRpcProvider,
    maxRetries = 3
  ): Promise<string> {
    const baseNonce = await provider.getTransactionCount(wallet.address, 'latest');

    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        const feeData = await provider.getFeeData();
        const populatedTx: TransactionRequest = {
          ...txRequest,
          gasLimit: txRequest.gasLimit || 500_000n,
          nonce: baseNonce + attempt,
          chainId: (await provider.getNetwork()).chainId,
        };

        // Use EIP-1559 if available, fallback to legacy
        if (feeData.maxFeePerGas) {
          populatedTx.maxFeePerGas = feeData.maxFeePerGas;
          populatedTx.maxPriorityFeePerGas = feeData.maxPriorityFeePerGas;
        } else {
          populatedTx.gasPrice = feeData.gasPrice;
        }

        const tx = await wallet.sendTransaction(populatedTx);
        log.info({ txHash: tx.hash, attempt: attempt + 1 }, 'TX sent (passthrough)');

        const receipt = await tx.wait(3);
        if (receipt && receipt.status === 1) {
          log.info({ block: receipt.blockNumber, txHash: tx.hash }, 'TX confirmed');
          return tx.hash;
        }
        throw new Error('TX failed or receipt is null');
      } catch (err) {
        log.warn({ attempt: attempt + 1, maxRetries, err }, 'Passthrough TX retry');
        if (attempt >= maxRetries) throw err;
      }
    }
    throw new Error('Passthrough max retries reached');
  }
}
```
