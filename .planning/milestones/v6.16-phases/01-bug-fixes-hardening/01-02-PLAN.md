**Modify `src/utils/flashbots.ts`:**

Replace the entire file content with:

```typescript
import { providers, Wallet } from "ethers";
import { FlashbotsBundleProvider } from "@flashbots/ethers-provider-bundle";
import { CONFIG, SupportedNetwork } from "../config";

const FLASHBOTS_RELAYS: Record<string, { url: string; network: string }> = {
    sepolia: { url: "https://relay-sepolia.flashbots.net", network: "sepolia" },
    mainnet: { url: "https://relay.flashbots.net", network: "mainnet" },
};

export async function getFlashbotsProvider(
    provider: providers.JsonRpcProvider,
    authSigner: Wallet
): Promise<FlashbotsBundleProvider> {
    const relay = FLASHBOTS_RELAYS[CONFIG.network];
    if (!relay) {
        throw new Error(`No Flashbots relay configured for network: ${CONFIG.network}. Supported: ${Object.keys(FLASHBOTS_RELAYS).join(", ")}`);
    }
    return await FlashbotsBundleProvider.create(
        provider,
        authSigner,
        relay.url,
        relay.network
    );
}
```

Key changes:
- Import `CONFIG` and `SupportedNetwork` from `"../config"`
- Create `FLASHBOTS_RELAYS` lookup map with sepolia (`https://relay-sepolia.flashbots.net`) and mainnet (`https://relay.flashbots.net`) entries
- Use `CONFIG.network` to look up the correct relay at runtime
- Throw explicit error with supported networks listed if `CONFIG.network` has no relay entry
- Remove hardcoded `"https://relay.flashbots.net"` and `"mainnet"` strings
