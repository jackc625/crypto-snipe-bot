  - On any RPC error: outcome='inconclusive', wethSaved=null.

```
async function resumeAntiRugMonitoring(contexts: ChainContext[]): Promise<void> {
    // Step 1: per-chain allowlist verification per D-LIST-VERIFY (one call per chain).
    for (const ctx of contexts) {
        try {
            await verifyAllowlistAtStartup(ctx, ctx.chain.antiRug.allowlistV2, ctx.chain.antiRug.allowlistV3);
        } catch (err) {
            // verifyAllowlistAtStartup process.exit(1) on EOA per D-LIST-VERIFY; if we reach here, RPC failed
            ctx.log.error({ err, chain: ctx.chain.name, op: 'antirug.allowlist-verify-rpc-failed' }, 'Allowlist verify RPC failed; continuing without anti-rug on this chain');
            continue;
        }
    }

    // Step 2: scan trades JOIN lp_baseline for open positions to re-register.
    const openTrades = db.prepare(`
        SELECT t.id, t.token, t.chain, t.pool_type, t.fee_tier, b.pair_or_pool_address
        FROM trades t
        INNER JOIN lp_baseline b ON b.trade_id = t.id
        WHERE t.status = 'pending' AND t.action = 'buy'
    `).all() as Array<{
        id: number;
        token: string;
        chain: string;
        pool_type: 'v2' | 'v3';
        fee_tier: number | null;
        pair_or_pool_address: string;
    }>;

    for (const trade of openTrades) {
        const ctx = contexts.find(c => c.chain.name === trade.chain);
        if (!ctx) continue;
        // Find the dex matching pool_type — first v2 dex for v2 positions, the v3 dex for v3
        const dex = ctx.chain.dexes.find(d => d.dexType === trade.pool_type);
        if (!dex) {
            ctx.log.error({ tradeId: trade.id, poolType: trade.pool_type, op: 'antirug.resume.no-dex' }, 'Cannot resume anti-rug — no matching dex');
            continue;
        }
        registerAntiRugPosition(ctx, dex, trade.token, trade.pair_or_pool_address, trade.id, trade.fee_tier);
    }

    // Step 3: kick off per-chain listener (registers heartbeat + backfill).
    for (const ctx of contexts) {
        await startChainAntiRugListener(ctx);
    }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| `executeFullSell` reason union | type-system boundary; an executor that calls executeFullSell with a stringified reason variant that isn't in the union must fail compilation, not silently take a wrong branch |
| `triggerAntiRugExit` dry-run gate | env-driven boundary; the `ctx.chain.antiRug.liveExecute` flag is the single gate between observability-only and actual capital movement |
