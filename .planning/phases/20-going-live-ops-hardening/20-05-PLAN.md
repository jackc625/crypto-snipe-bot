       // scripts/preflight.ts
       // Phase 20 / Plan 20-05 / OPS-LIVE-10 — operator CLI wrapping preflightDryRun().
       // Runs all OPS-LIVE-02..07 checks + prints per-chain pass/fail summary.
       // Exits 0 on all-pass / 1 on any FATAL / 2 on uncaught throw.
       // Does NOT start the bot.
       import * as dotenv from 'dotenv';
       import { buildChainRegistry } from '../src/chainRegistry';
       import { preflightDryRun } from '../src/preflight';
       import type { PreflightResult } from '../src/preflight';
       import { logger } from '../src/utils/logger';

       dotenv.config();
       const log = logger.child({ module: 'preflight-cli' });

       function printHumanSummary(results: PreflightResult[]): void {
           // Group by chain, print ✓/✗ per check
           const byChain = new Map<string, PreflightResult[]>();
           for (const r of results) {
               if (!byChain.has(r.chain)) byChain.set(r.chain, []);
               byChain.get(r.chain)!.push(r);
           }
           for (const [chain, chainResults] of byChain) {
               log.info({ chain }, `--- ${chain.toUpperCase()} ---`);
               for (const r of chainResults) {
                   const symbol = r.passed ? '✓' : '✗';
                   log.info({ chain, check: r.check, passed: r.passed, reason: r.reason }, `  ${symbol} ${r.check}${r.passed ? '' : ': ' + r.reason}`);
               }
           }
       }

       async function main(): Promise<void> {
           const useJson = process.argv.includes('--json');
           const registry = buildChainRegistry();
           const results = await preflightDryRun(registry);
           const failed = results.filter(r => !r.passed);
           if (useJson) {
               console.log(JSON.stringify({ results, failed: failed.length }, null, 2));
           } else {
               printHumanSummary(results);
               if (failed.length > 0) {
                   log.fatal({ failedCount: failed.length }, `Preflight FAILED (${failed.length} check${failed.length === 1 ? '' : 's'} failed across ${results.length} total)`);
               } else {
                   log.info({ totalChecks: results.length }, `Preflight PASSED (${results.length} checks across ${new Set(results.map(r => r.chain)).size} chains)`);
               }
           }
           process.exit(failed.length > 0 ? 1 : 0);
       }

       main().catch((err) => {
           log.fatal({ err: err instanceof Error ? err.message : String(err) }, 'Preflight CLI crashed');
           process.exit(2);
       });
       ```

       // scripts/pause.ts
       // Phase 20 / Plan 20-05 / OPS-LIVE-11 — operator CLI to pause the bot via dashboard endpoint.
       // Two-step flow: POST /api/auth/login with DASHBOARD_API_KEY → cookie-forwarded POST /api/controls/pause.
       // Exits 0 on success / 1 on HTTP error / 2 on uncaught throw.
       // T-20-05 mitigation: API key passed via env (process.env.DASHBOARD_API_KEY) and request body — NEVER logged.
       import * as dotenv from 'dotenv';
       import axios from 'axios';
       import { logger } from '../src/utils/logger';

       dotenv.config();
       const log = logger.child({ module: 'pause-cli' });

       async function main(): Promise<void> {
           const apiKey = process.env.DASHBOARD_API_KEY;
           const dashboardUrl = process.env.DASHBOARD_URL || 'http://localhost:3000';
           const reasonArgIdx = process.argv.indexOf('--reason');
           const reason = reasonArgIdx >= 0 && reasonArgIdx + 1 < process.argv.length ? process.argv[reasonArgIdx + 1] : 'CLI pause';

           if (!apiKey) {
               log.fatal('DASHBOARD_API_KEY not set in .env');
               process.exit(1);
           }

           // Step 1: login (sends apiKey in request body; NEVER log the apiKey value)
           const loginRes = await axios.post(
               `${dashboardUrl}/api/auth/login`,
               { apiKey },
               { withCredentials: true, validateStatus: () => true },
           );
           if (loginRes.status !== 200) {
               log.fatal({ status: loginRes.status, body: loginRes.data }, 'Login failed');
               process.exit(1);
           }
           const cookie = loginRes.headers['set-cookie']?.[0] || '';

           // Step 2: POST /pause with forwarded cookie
           const pauseRes = await axios.post(
               `${dashboardUrl}/api/controls/pause`,
               { reason, pausedBy: 'cli' },
               { headers: { Cookie: cookie }, validateStatus: () => true },
           );
           if (pauseRes.status !== 200) {
               log.fatal({ status: pauseRes.status, body: pauseRes.data }, 'Pause failed');
               process.exit(1);
           }
           log.info({ reason, response: pauseRes.data }, 'Bot paused via CLI');
           process.exit(0);
       }

       main().catch((err) => {
           log.fatal({ err: err instanceof Error ? err.message : String(err) }, 'Pause CLI crashed');
           process.exit(2);
       });
       ```
