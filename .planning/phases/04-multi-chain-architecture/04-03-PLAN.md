**`src/utils/safetyProviders/honeypotIs.ts`:**

Extract existing honeypot.is API logic from tokenSafety.ts. Make chain-aware by using `ctx.chain.chainId` to set the correct API chain parameter.

```typescript
import axios from 'axios';
import { retry } from '../retry';
import { logger } from '../logger';
import type { ChainContext } from '../../chainRegistry';

const log = logger.child({ module: 'safety-honeypotIs' });

// honeypot.is supports: ETH (chainId 1), BSC (chainId 56), Base (chainId 8453)
// API: GET https://api.honeypot.is/v2/IsHoneypot?address={token}&chainID={chainId}
export async function honeypotIsCheck(ctx: ChainContext, tokenAddress: string): Promise<boolean> {
  try {
    const { data } = await retry(() =>
      axios.get('https://api.honeypot.is/v2/IsHoneypot', {
        params: {
          address: tokenAddress,
          chainID: ctx.chain.chainId
        },
        timeout: 10000
      })
    );

    if (data.honeypotResult) {
      return data.honeypotResult.isHoneypot === true;
    }

    // Fallback to v1 format
    return data.isHoneypot === true;
  } catch (err) {
    ctx.log.warn({ err, token: tokenAddress, check: 'honeypot.is' }, 'honeypot.is API failed');
    return true; // default to blocking on API failure
  }
}
```

**`src/utils/safetyProviders/goplus.ts`:**

GoPlus Security API adapter. Covers ALL four chains. Per research: `GET https://api.gopluslabs.io/api/v1/token_security/{chain_id}?contract_addresses={address}`. No API key needed for basic use.

```typescript
import axios from 'axios';
import { retry } from '../retry';
import { logger } from '../logger';
import type { ChainContext } from '../../chainRegistry';

const log = logger.child({ module: 'safety-goplus' });

const GOPLUS_BASE_URL = 'https://api.gopluslabs.io/api/v1';

interface GoPlusTokenResult {
  is_honeypot?: string;          // '0' or '1'
  buy_tax?: string;              // percentage string
  sell_tax?: string;             // percentage string
  is_open_source?: string;       // '0' or '1'
  is_proxy?: string;             // '0' or '1'
  lp_holders?: Array<{
    address: string;
    is_locked: number;           // 0 or 1
    percent: number;
    locked_detail?: Array<{ end_time: string }>;
  }>;
}

async function fetchGoPlusTokenSecurity(chainId: number, tokenAddress: string): Promise<GoPlusTokenResult | null> {
  try {
    const { data } = await retry(() =>
      axios.get(`${GOPLUS_BASE_URL}/token_security/${chainId}`, {
        params: { contract_addresses: tokenAddress },
        timeout: 10000
      })
    );

    if (data.code !== 1 || !data.result) {
      log.warn({ chainId, token: tokenAddress }, 'GoPlus returned no data');
      return null;
    }

    return data.result[tokenAddress.toLowerCase()] || null;
  } catch (err) {
    log.warn({ err, chainId, token: tokenAddress }, 'GoPlus API request failed');
    return null;
  }
}

export async function goplusHoneypotCheck(ctx: ChainContext, tokenAddress: string): Promise<boolean> {
  const result = await fetchGoPlusTokenSecurity(ctx.chain.chainId, tokenAddress);
  if (!result) return true; // default to blocking if API fails
  return result.is_honeypot === '1';
}

export async function goplusLiquidityLockCheck(ctx: ChainContext, pairAddress: string): Promise<boolean> {
  // GoPlus checks LP holder lock status
  const result = await fetchGoPlusTokenSecurity(ctx.chain.chainId, pairAddress);
  if (!result || !result.lp_holders) return false; // no lock info = not locked

  const now = Math.floor(Date.now() / 1000);
  return result.lp_holders.some(holder => {
    if (holder.is_locked !== 1) return false;
    // Check if lock is still active
    if (holder.locked_detail && holder.locked_detail.length > 0) {
      return holder.locked_detail.some(detail => parseInt(detail.end_time) > now);
    }
    return holder.is_locked === 1;
  });
}

export async function goplusTokenSecurity(ctx: ChainContext, tokenAddress: string): Promise<GoPlusTokenResult | null> {
  return fetchGoPlusTokenSecurity(ctx.chain.chainId, tokenAddress);
}
```

**`src/utils/safetyProviders/unicrypt.ts`:**

Extract existing Unicrypt liquidity lock check from tokenSafety.ts. Pass chainId to API.

```typescript
import axios from 'axios';
import { Contract } from 'ethers';
import { retry } from '../retry';
import { logger } from '../logger';
import { PAIR_ABI } from '../abis';
import type { ChainContext } from '../../chainRegistry';

const log = logger.child({ module: 'safety-unicrypt' });
const UNICRYPT_API = 'https://api.unicrypt.network/api/v2/locker/pair/';

export async function unicryptLockCheck(ctx: ChainContext, pairAddress: string): Promise<boolean> {
  try {
    const { data } = await retry(() =>
      axios.get(`${UNICRYPT_API}${pairAddress}?chainId=${ctx.chain.chainId}`)
    );
    const locks = data.locks;

    if (!locks || locks.length === 0) {
      ctx.log.warn({ pair: pairAddress, check: 'unicrypt' }, 'No liquidity lock found');
      return false;
    }

    const now = Date.now() / 1000;
    return locks.some((lock: any) => lock.unlock_date > now);
  } catch (e) {
    ctx.log.warn({ pair: pairAddress, check: 'unicrypt', fallback: 'burnCheck' }, 'Unicrypt API failed, falling back to burn check');
    return await fallbackCheckBurned(ctx, pairAddress);
  }
}

async function fallbackCheckBurned(ctx: ChainContext, pairAddress: string): Promise<boolean> {
  try {
    const pair = new Contract(pairAddress, PAIR_ABI, ctx.httpProvider);
    const burned = await pair.balanceOf('0x000000000000000000000000000000000000dEaD');
    const total = await pair.totalSupply();
    const percentBurned = Number(burned) / Number(total);

    if (percentBurned > 0.9) {
      ctx.log.info({ pair: pairAddress, percentBurned: (percentBurned * 100).toFixed(1) }, 'LP appears burned');
      return true;
    }
    return false;
  } catch (e) {
    ctx.log.warn({ err: e, check: 'burnFallback' }, 'Burn fallback failed');
    return false;
  }
}
```

**`src/utils/safetyProviders/mudra.ts`:**

BSC liquidity lock checker via Mudra Locker API.

```typescript
import axios from 'axios';
import { retry } from '../retry';
import { logger } from '../logger';
import type { ChainContext } from '../../chainRegistry';

const log = logger.child({ module: 'safety-mudra' });

// Mudra Locker API for BSC
// Endpoint: GET https://api.mudra.website/api/v1/locker/pair/{pairAddress}
export async function mudraLockCheck(ctx: ChainContext, pairAddress: string): Promise<boolean> {
  try {
    const { data } = await retry(() =>
      axios.get(`https://api.mudra.website/api/v1/locker/pair/${pairAddress}`, {
        timeout: 10000
      })
    );

    if (!data || !data.data || !data.data.locks || data.data.locks.length === 0) {
      ctx.log.warn({ pair: pairAddress, check: 'mudra' }, 'No Mudra lock found');
      return false;
    }

    const now = Math.floor(Date.now() / 1000);
    return data.data.locks.some((lock: any) => lock.unlockDate > now);
  } catch (e) {
    ctx.log.warn({ err: e, pair: pairAddress, check: 'mudra' }, 'Mudra API failed');
    // Fallback to GoPlus for BSC liquidity lock check
    try {
      const { goplusLiquidityLockCheck } = await import('./goplus');
      return await goplusLiquidityLockCheck(ctx, pairAddress);
    } catch (fallbackErr) {
      ctx.log.warn({ err: fallbackErr, pair: pairAddress, check: 'mudra-fallback' }, 'GoPlus fallback also failed');
      return false;
    }
  }
}
```
