```typescript
export function applyGasMultiplier(fee, gasMultiplier) {
  if (fee === null || fee === undefined) return fee;
  if (gasMultiplier === undefined || gasMultiplier === 1) return fee;
  return fee * BigInt(Math.round(gasMultiplier * 100)) / 100n;
}
```
