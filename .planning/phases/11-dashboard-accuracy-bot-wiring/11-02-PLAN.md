Replace the entire GET / handler body. The route factory signature stays the same: `positionRoutes(contexts: ChainContext[]): Router`.

New implementation:
```typescript
import { Router } from 'express';
import type { Request, Response } from 'express';
import type { ChainContext } from '../../chainRegistry';
import { buildPositionsWithLivePrices } from '../helpers/positions';

function asyncHandler(fn: (req: Request, res: Response) => Promise<void>) {
    return (req: Request, res: Response, next: (err?: unknown) => void) => {
        fn(req, res).catch(next);
    };
}

export function positionRoutes(contexts: ChainContext[]): Router {
    const router = Router();

    router.get('/', asyncHandler(async (_req: Request, res: Response) => {
        const positions = await buildPositionsWithLivePrices(contexts);
        res.json({ positions });
    }));

    return router;
}
```

```typescript
router.post('/:name/start', async (req: Request, res: Response) => {
    const name = String(req.params.name).toLowerCase();
    const ctx = contexts.find((c) => c.chain.name.toLowerCase() === name);

    if (!ctx) {
        res.status(404).json({ error: 'Chain not found' });
        return;
    }

    if (!stoppedChains.has(ctx.chain.name)) {
        res.status(400).json({ error: 'Chain is already running' });
        return;
    }

    stoppedChains.delete(ctx.chain.name);

    if (onRestart) {
        try {
            const result = await onRestart(ctx);
            ctx.log.info({ warnings: result.warnings }, 'Chain listeners restarted via dashboard');
            res.json({
                ok: true,
                chain: ctx.chain.name,
                status: 'running',
                warnings: result.warnings,  // D-12: always include warnings array
            });
        } catch (err) {
            ctx.log.error({ err }, 'Failed to restart chain listeners');
            // Even on full failure, chain was removed from stopped set
            res.json({
                ok: true,
                chain: ctx.chain.name,
                status: 'running',
                warnings: [`Full restart failed: ${err instanceof Error ? err.message : String(err)}`],
            });
        }
    } else {
        ctx.log.info('Chain removed from stopped set via dashboard (no restart callback)');
        res.json({
            ok: true,
            chain: ctx.chain.name,
            status: 'running',
            warnings: ['No restart callback configured. Full listener restart requires bot restart.'],
        });
    }
});
```

   ```typescript
   const onRestart = async (ctx: ChainContext): Promise<{ warnings: string[] }> => {
       const warnings: string[] = [];

       // D-11: Reconnect WebSocket provider if disconnected
       if (!ctx.wsProvider || ctx.wsProvider.destroyed) {
           if (ctx.chain.wsRpcUrl) {
               removeWsProvider(ctx.chain.name);
               const newWs = getWsProvider(ctx.chain.name, ctx.chain.wsRpcUrl, ctx.chain.chainId);
               ctx.wsProvider = newWs;
               ctx.log.info('WebSocket provider reconnected for chain restart');
           } else {
               warnings.push('No WebSocket URL configured, using HTTP polling');
           }
       }

       // D-10: MEV health check (warn only, don't block)
       try {
           const strategy = createMevStrategy(ctx.chain);
           const mevHealthy = await strategy.healthCheck();
           if (!mevHealthy) {
               warnings.push('MEV service unreachable');
               ctx.log.warn('MEV service unreachable during chain restart');
           }
       } catch (err) {
           warnings.push(`MEV health check failed: ${err instanceof Error ? err.message : String(err)}`);
       }

       // Re-attach listeners per DEX (D-12: collect per-DEX results)
       for (const dex of ctx.chain.dexes) {
           try {

               warnings.push(msg);
               ctx.log.error({ err, dex: dex.name }, 'Failed to restart DEX listener');
           }
   ```
