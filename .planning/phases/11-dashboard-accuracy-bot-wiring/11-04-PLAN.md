        const senderId = '999';
        const chatId = '888';

        const senderAllowed = allowedSenders.size === 0 || allowedSenders.has(senderId);
        const channelAllowed = allowedChannels.size === 0 || allowedChannels.has(chatId);

        // With empty sets, both checks pass (size === 0 short-circuits)
        expect(senderAllowed).toBe(true);
        expect(channelAllowed).toBe(true);
        // Message should be allowed: !(false) since both are true = at least one must be true
        expect(senderAllowed || channelAllowed).toBe(true);

        const allowedSenders = new Set(['111', '222']);

        const chatId = '888';

        const senderAllowed = allowedSenders.size === 0 || allowedSenders.has(senderId);
        const channelAllowed = allowedChannels.size === 0 || allowedChannels.has(chatId);

        expect(senderAllowed).toBe(false);
        expect(channelAllowed).toBe(false);
        // Message rejected: neither sender nor channel is allowed
        expect(!senderAllowed && !channelAllowed).toBe(true);

    it('whitelist allows when sender is whitelisted but channel is not', () => {
        const allowedSenders = new Set(['111']);
        const allowedChannels = new Set(['333']);

        const chatId = '888';

        const senderAllowed = allowedSenders.size === 0 || allowedSenders.has(senderId);
        const channelAllowed = allowedChannels.size === 0 || allowedChannels.has(chatId);

        expect(senderAllowed).toBe(true);
        expect(channelAllowed).toBe(false);
        // OR logic: message allowed if either passes
        expect(!(!senderAllowed && !channelAllowed)).toBe(true);
    });
});
